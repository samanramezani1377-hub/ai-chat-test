cmake_minimum_required(VERSION 3.22)
project(ai_chat_mnn_runtime LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(FETCHCONTENT_QUIET OFF)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "Use pinned MNN source without implicit updates")
set(MNN_TAG "d407447ed56c4121a11ccbd266dc184ca1ead0c2" CACHE STRING "Pinned MNN 3.6.1 commit")
set(MNN_BUILD_LLM ON CACHE BOOL "" FORCE)
set(MNN_OPENCL ON CACHE BOOL "" FORCE)
set(MNN_ARM82 ON CACHE BOOL "" FORCE)
set(MNN_LOW_MEMORY ON CACHE BOOL "" FORCE)
set(MNN_SUPPORT_TRANSFORMER_FUSE ON CACHE BOOL "" FORCE)
set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_USE_LOGCAT ON CACHE BOOL "" FORCE)
set(MNN_BUILD_TRAIN OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_CONVERTER OFF CACHE BOOL "" FORCE)
FetchContent_Declare(mnn GIT_REPOSITORY https://github.com/alibaba/MNN.git GIT_TAG ${MNN_TAG} GIT_SHALLOW TRUE GIT_SUBMODULES_RECURSE TRUE)
FetchContent_MakeAvailable(mnn)
if (NOT TARGET llm)
    message(FATAL_ERROR "Pinned MNN source did not expose the LLM target. Check MNN_BUILD_LLM and upstream target names.")
endif()
# The upstream LLM engine only accumulates Llm::getLog() when this is defined
# while compiling llm.cpp. This turns model-load failures into actionable diagnostics.
target_compile_definitions(llm PRIVATE LLM_LOG_TO_STRING)
add_library(ai_chat_mnn_runtime SHARED mnn_runtime.cpp)
target_include_directories(ai_chat_mnn_runtime PRIVATE "${mnn_SOURCE_DIR}/include" "${mnn_SOURCE_DIR}/transformers/llm/engine/include")
target_compile_options(ai_chat_mnn_runtime PRIVATE -fexceptions -frtti)
target_link_libraries(ai_chat_mnn_runtime PRIVATE llm MNN_Express MNN android log)
